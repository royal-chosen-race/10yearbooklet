        function moveEntry(id, direction) {

            const idx = state.entries.findIndex(e => e.id === id);

            if (direction === 'up' && idx > 0) {

                [state.entries[idx], state.entries[idx-1]] = [state.entries[idx-1], state.entries[idx]];

            } else if (direction === 'down' && idx < state.entries.length - 1) {

                [state.entries[idx], state.entries[idx+1]] = [state.entries[idx+1], state.entries[idx]];

            }

            saveState(); render();

        }

        // ── 智慧行政快捷工作流按鈕組 ──

        function duplicateEntry(id, mode) {

            const idx = state.entries.findIndex(e => e.id === id);

            if (idx === -1) return;

            const source = state.entries[idx];

            const newId = Date.now().toString();

            let newEntry = {};


            if (mode === 'clone') {

                newEntry = { ...source, id: newId, name: source.name + ' (副本)' };

            } else if (mode === 'same_group') {

                newEntry = {

                    id: newId, layout: source.layout, groupName: source.groupName,

                    name: '新肢體成員', role: source.role, title: source.title, headline: '', content: ''

                };

            }

            state.entries.splice(idx + 1, 0, newEntry);

            state.activeId = newId; saveState(); render();

        }

            }

        }

        // ── 核心：真實現體高度排版與落頁引擎 (True-Height DOM Engine) ──

        function renderCanvas() {

            const container = document.getElementById('canvas');

            const pc = document.getElementById('printArea');

            container.innerHTML = ''; pc.innerHTML = '';

            

            let scratchpad = document.createElement('div');

            scratchpad.style.cssText = "position: absolute; visibility: hidden; width: 210mm; top: -9999px; left: -9999px; z-index:-100;";

            document.body.appendChild(scratchpad);


            // 統計各小組的人數，用於執行「少於3人小組扉頁收斂優化」

            let groupStats = {};

            state.entries.forEach(e => { groupStats[e.groupName] = (groupStats[e.groupName] || 0) + 1; });


            let initialFlattenedEntries = [];

            

            // 拆文演算法 (Paragraph Split Engine)

            state.entries.forEach(entry => {

                let paragraphs = entry.content.split('\n').filter(p => p.trim() !== '');

                let testCard = document.createElement('div');

                testCard.style.width = "85mm"; 

                testCard.innerHTML = buildCardHTMLString(entry, false);

                scratchpad.appendChild(testCard);

                let cardHeight = testCard.getBoundingClientRect().height;

                scratchpad.removeChild(testCard);


                if (cardHeight > 390 && paragraphs.length > 1 && entry.layout !== 'PASTOR') {

                    let chunk1 = []; let chunk2 = [];

                    let mid = Math.ceil(paragraphs.length / 2);

                    for(let i=0; i<paragraphs.length; i++) {

                        if(i < mid) chunk1.push(paragraphs[i]); else chunk2.push(paragraphs[i]);

                    }

                    initialFlattenedEntries.push({ ...entry, content: chunk1.join('\n'), headline: entry.headline ? `${entry.headline} (轉下頁)` : '(接續轉下頁)' });

                    initialFlattenedEntries.push({ ...entry, id: entry.id + '_cont', content: chunk2.join('\n'), headline: entry.headline ? `(接續) ${entry.headline}` : '(接續上頁)' });

                } else {

                    initialFlattenedEntries.push(entry);

                }

            });


            // 智慧對稱配對 (Dynamic Height Pairing) 

            let sortedFlowItems = []; let pastorItems = [];

            initialFlattenedEntries.forEach(e => {

                if(e.layout === 'PASTOR') pastorItems.push(e); else sortedFlowItems.push(e);

            });

            sortedFlowItems.sort((a, b) => b.content.length - a.content.length);

            let balancedFlowItems = [];

            while(sortedFlowItems.length > 0) {

                balancedFlowItems.push(sortedFlowItems.shift());

                if(sortedFlowItems.length > 0) balancedFlowItems.push(sortedFlowItems.pop());

            }


            let finalQueue = [...pastorItems, ...balancedFlowItems];

            let pages = []; let currentFlowItems = []; let lastGroup = "";

            const MAX_PRINTABLE_HEIGHT = 810; 


            finalQueue.forEach((entry) => {

                // 智慧分流判定：當前小組總人數 >= 3 人，才允許插入大扉頁；少於3人則改用 Inline Section Header

                if (entry.layout !== 'PASTOR' && entry.groupName !== lastGroup) {

                    if (currentFlowItems.length > 0) { pages.push({ type: 'FLOW', items: currentFlowItems }); currentFlowItems = []; }

                    

                    let count = groupStats[entry.groupName] || 0;

                    if (count >= 3) {

                        pages.push({ type: 'CHAPTER', groupName: entry.groupName });

                    } else {

                        // 標記該組的第一個人，在落頁時生成 Inline 標頭飾帶

                        entry.isFirstOfInlineSection = true;

                    }

                    lastGroup = entry.groupName;

                }


                if (entry.layout === 'PASTOR') {

                    if (currentFlowItems.length > 0) { pages.push({ type: 'FLOW', items: currentFlowItems }); currentFlowItems = []; }

                    pages.push({ type: 'PASTOR', items: [entry] });

                } else {

                    let testItems = [...currentFlowItems, entry];

                    let testPage = document.createElement('div'); testPage.className = "book-page";

                    let whiteLayer = document.createElement('div'); whiteLayer.className = "white-layer";

                    let gridEngine = document.createElement('div'); gridEngine.className = "grid-flow-engine pt-2";

                    

                    let rows = [];

                    for(let r = 0; r < testItems.length; r += 2) { rows.push([testItems[r], testItems[r+1]]); }

                    gridEngine.innerHTML = rows.map(row => `

                        <div class="flex gap-[14mm] mb-[20px] items-stretch w-full">

                            ${buildCardHTMLString(row[0], row[0]?.isFirstOfInlineSection)}

                            ${row[1] ? buildCardHTMLString(row[1], row[1]?.isFirstOfInlineSection) : '<div class="flex-1"></div>'}

                        </div>

                    `).join('');

                    

                    whiteLayer.appendChild(gridEngine); testPage.appendChild(whiteLayer); scratchpad.appendChild(testPage);

                    let measuredHeight = gridEngine.getBoundingClientRect().height;

                    scratchpad.removeChild(testPage);

                    

                    if (measuredHeight > MAX_PRINTABLE_HEIGHT && currentFlowItems.length > 0) {

                        pages.push({ type: 'FLOW', items: currentFlowItems });

                        currentFlowItems = [entry];

                    } else {

                        currentFlowItems.push(entry);

                    }

                }

            });

            if (currentFlowItems.length > 0) pages.push({ type: 'FLOW', items: currentFlowItems });

            document.body.removeChild(scratchpad);


            // ── 正統奇數在右、偶數在左對開落頁調度 ──

            let pageNum = 7; // 全書起始頁碼由 P.7 (右頁) 開始

            

            if (state.viewMode === 'spread') {

                // 正統出版規範：P.7 為單獨右頁，前方自動補空左頁（P.6）以對齊印刷排線

                const firstSpread = document.createElement('div');

                firstSpread.className = "flex shadow-2xl mb-12";

                firstSpread.innerHTML = `<div class="book-page bg-slate-950/40 border border-dashed border-white/5 flex items-center justify-center text-white/20 text-xs font-mono">（前頁 Verso 留白）</div><div class="w-[1.5px] bg-gray-400/30 z-20 no-print"></div>` + buildPageHTML(pages[0], pageNum);

                firstSpread.querySelectorAll('.book-page').forEach(el => { el.style.boxShadow = 'none'; el.style.marginBottom = '0'; });

                container.appendChild(firstSpread);

                

                // 後續頁面進入 2x2 標準對開配對

                let i = 1;

                pageNum++; // 進入 P.8 (左頁)

                while (i < pages.length) {

                    const spread = document.createElement('div');

                    spread.className = "flex shadow-2xl mb-12";

                    

                    let leftPageHtml = buildPageHTML(pages[i], pageNum); pageNum++; i++;

                    let rightPageHtml = (i < pages.length) ? buildPageHTML(pages[i], pageNum) : `<div class="book-page bg-slate-950/40"></div>`;

                    pageNum++; i++;

                    

                    spread.innerHTML = leftPageHtml + `<div class="w-[1.5px] bg-gray-400/30 z-20 no-print"></div>` + rightPageHtml;

                    spread.querySelectorAll('.book-page').forEach(el => { el.style.boxShadow = 'none'; el.style.marginBottom = '0'; });

                    container.appendChild(spread);

                }

            } else {

                pages.forEach(p => {

                    const div = document.createElement('div'); div.className = "mb-12 shadow-2xl";

                    div.innerHTML = buildPageHTML(p, pageNum); div.querySelector('.book-page').style.boxShadow = 'none';

                    container.appendChild(div); pageNum++;

                });

            }


            // 完全同步到印刷快取

            let pNumCache = 7;

            pc.innerHTML = pages.map(p => { let html = buildPageHTML(p, pNumCache); pNumCache++; return html; }).join('');

            pc.querySelectorAll('.book-page').forEach(el => el.style.boxShadow = 'none');


            // 最終防呆溢出紅框檢驗

            setTimeout(validateOverflowGuardrails, 300);

        }
